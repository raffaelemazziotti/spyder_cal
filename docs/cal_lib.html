 15def _bundled_libusb_backend():
 16    """Return a backend for the libusb DLL shipped with ``libusb-package``.
 17
 18    Windows has no system libusb, so the automatic discovery normally fails
 19    there. ``libusb-package`` (pip-installable) bundles ``libusb-1.0.dll``;
 20    returns None when it is not installed.
 21    """
 22    try:
 23        import libusb_package
 24    except ImportError:
 25        return None
 26    library_path = libusb_package.get_library_path()
 27    if library_path is None:
 28        return None
 29    return usb.backend.libusb1.get_backend(
 30        find_library=lambda _name: str(library_path)
 31    )

 34class GammaMeasurement:
 35    """Result of one or more display-gamma measurement runs."""
 36
 37    def __init__(self, fits):
 38        if not fits:
 39            raise ValueError("at least one gamma fit is required")
 40
 41        self.runs = []
 42        for index, fit in enumerate(fits, start=1):
 43            gray_levels = np.asarray(fit.original_intensities, dtype=float)
 44            luminance = np.asarray(fit.original_luminance, dtype=float)
 45            self.runs.append(
 46                {
 47                    "repetition": index,
 48                    "gamma": float(fit.gamma),
 49                    "gray_levels": gray_levels.tolist(),
 50                    "luminance": luminance.tolist(),
 51                    "fit_parameters": np.asarray(fit.params, dtype=float).tolist(),
 52                }
 53            )
 54
 55        self.gamma_values = [run["gamma"] for run in self.runs]
 56        self.gamma = fmean(self.gamma_values)
 57        self.gamma_std = (
 58            stdev(self.gamma_values) if len(self.gamma_values) > 1 else 0.0
 59        )
 60        self.luminance_min_values = [min(run["luminance"]) for run in self.runs]
 61        self.luminance_max_values = [max(run["luminance"]) for run in self.runs]
 62        self.luminance_min = fmean(self.luminance_min_values)
 63        self.luminance_max = fmean(self.luminance_max_values)
 64        self.luminance_min_std = (
 65            stdev(self.luminance_min_values)
 66            if len(self.luminance_min_values) > 1
 67            else 0.0
 68        )
 69        self.luminance_max_std = (
 70            stdev(self.luminance_max_values)
 71            if len(self.luminance_max_values) > 1
 72            else 0.0
 73        )
 74
 75    def to_dict(self):
 76        """Return a JSON-serializable calibration result."""
 77        return {
 78            "schema_version": 2,
 79            "gamma": self.gamma,
 80            "gamma_std": self.gamma_std,
 81            "gamma_by_repetition": self.gamma_values,
 82            "luminance": {
 83                "unit": "cd/m^2",
 84                "minimum": self.luminance_min,
 85                "maximum": self.luminance_max,
 86                "minimum_std": self.luminance_min_std,
 87                "maximum_std": self.luminance_max_std,
 88                "minimum_by_repetition": self.luminance_min_values,
 89                "maximum_by_repetition": self.luminance_max_values,
 90            },
 91            "runs": self.runs,
 92        }
 93
 94    def save_json(self, path):
 95        """Atomically save this measurement and return its absolute path."""
 96        output_path = Path(path).expanduser().resolve()
 97        output_path.parent.mkdir(parents=True, exist_ok=True)
 98        temporary_path = output_path.with_suffix(output_path.suffix + ".tmp")
 99        temporary_path.write_text(
100            json.dumps(self.to_dict(), indent=2) + "\n",
101            encoding="utf-8",
102        )
103        temporary_path.replace(output_path)
104        return output_path

106    def __str__(self):
107        return (

109            f"luminance={self.luminance_min:.3f}–"
110            f"{self.luminance_max:.3f} cd/m²"

115    """
116    SpyderX colorimeter using PyUSB and a libusb 1.x backend.
117    """
118
119    INTERFACE = 0
120    VENDOR_ID = 0x085C
121    PRODUCT_ID = 0x0A00
122
123    def __init__(self, libusb_path=None):
124        """
125        Initialize the SpyderX device.
126
127        Args:
128            libusb_path (str, optional): Explicit path to a libusb 1.x shared
129                library. This preserves the existing Windows DLL workflow. If
130                omitted, PyUSB discovers the system libusb automatically and
131                falls back to the DLL bundled with ``libusb-package``.
132        """
133        self.backend = None
134        self.dev = None
135        self._interface_claimed = False
136        self._detached_kernel_driver = False
137        self._closed = False
138
139        if libusb_path is None:
140            self.backend = usb.backend.libusb1.get_backend()
141            if self.backend is None:
142                self.backend = _bundled_libusb_backend()
143        else:
144            self.backend = usb.backend.libusb1.get_backend(
145                find_library=lambda _name: libusb_path
146            )
147
148        if self.backend is None:
149            if libusb_path is None:
150                raise RuntimeError(
151                    "PyUSB could not find a libusb 1.x backend. Install libusb-1.0 "
152                    "and make it discoverable by the operating system (on "
153                    "Windows: pip install libusb-package), or pass libusb_path "
154                    "explicitly."
155                )
156            raise RuntimeError(
157                f"PyUSB could not load the libusb 1.x library at: {libusb_path}"
158            )
159
160        # Find SpyderX device on the bus
161        self.dev = usb.core.find(
162            idVendor=self.VENDOR_ID,
163            idProduct=self.PRODUCT_ID,
164            backend=self.backend,
165        )
166        if self.dev is None:
167            raise ValueError(
168                "SpyderX device (USB 085c:0a00) not found. Check that it is "
169                "connected, that your user has USB permission, and on Windows "
170                "that a libusb-compatible driver is installed."
171            )
172
173        try:
174            # An already configured device can reject this harmlessly.
175            try:
176                self.dev.set_configuration()
177            except usb.core.USBError:
178                pass
179
180            self._detach_kernel_driver_if_needed()
181            try:
182                usb.util.claim_interface(self.dev, self.INTERFACE)
183            except usb.core.USBError as error:
184                if platform.system() == "Linux" and getattr(error, "errno", None) == 13:
185                    raise PermissionError(
186                        "Linux denied access to SpyderX USB 085c:0a00. Install "
187                        "udev/60-spyderx.rules, reload udev rules, and unplug/"
188                        "reconnect the device. Do not run as root."
189                    ) from error
190                raise
191            self._interface_claimed = True
192
193            # Initialize the device (control transfers, then get calibration data)
194            self._initialize_device()
195        except Exception:
196            self.close()
197            raise
198
199    def _detach_kernel_driver_if_needed(self):
200        """Detach an active kernel driver when the backend supports doing so."""
201        is_active = getattr(self.dev, "is_kernel_driver_active", None)
202        detach = getattr(self.dev, "detach_kernel_driver", None)
203        if not callable(is_active) or not callable(detach):
204            return
205
206        try:
207            if is_active(self.INTERFACE):
208                detach(self.INTERFACE)
209                self._detached_kernel_driver = True
210        except (NotImplementedError, usb.core.USBError):
211            # Windows and some libusb backends do not implement these methods.
212            pass
213
214    def _initialize_device(self):
215        """
216        Send the standard control transfers, then retrieve factory calibration and measurement setup.
217        """
218        # Standard set of control transfers to 'wake' the SpyderX
219        self._ctrl(0x02, 1, 0, 1)
220        self._ctrl(0x02, 1, 0, 129)
221        self._ctrl(0x41, 2, 2, 0)
222
223        # Retrieve factory calibration (matrix, v1, v2, etc.)
224        self._get_calibration()
225        # Retrieve measurement setup values (s1, s2, s3)
226        self._setup_measurement()
227
228    def _ctrl(self, bmRequestType, bRequest, wValue, wIndex):
229        """
230        Helper for control transfers.
231        """
232        self.dev.ctrl_transfer(bmRequestType, bRequest, wValue, wIndex, None)
233
234    def _bulk(self, data, read_size, timeout=1000):
235        """
236        Helper for bulk transfers. Writes 'data' to endpoint 1, reads 'read_size' bytes from endpoint 0x81.
237        """
238        self.dev.write(1, data, timeout=timeout)
239        return self.dev.read(0x81, read_size, timeout=timeout)
240
241    def _read_ieee754(self, b):
242        """
243        Decode a 32-bit IEEE-754 float from 4 bytes in 'b'.
244        The SpyderX data is little-endian, so we reverse and parse.
245        """
246        b = b[::-1]
247        raw = int.from_bytes(b, "big")
248        sign = (raw >> 31) & 1
249        exponent = (raw >> 23) & 0xFF
250        fraction = raw & 0x7FFFFF
251        return (-1) ** sign * (1 + fraction / 2**23) * 2**(exponent - 127)
252
253    def _get_calibration(self):
254        """
255        Retrieve factory calibration data, storing:
256          - v1, v2: calibration parameters
257          - calibration_matrix: 3x3 for XYZ computation
258        """
259        # Command 0xcb for factory calibration, read 47 bytes
260        out = self._bulk([0xcb, 0x05, 0x73, 0x00, 0x01, 0x00], 47)[5:]
261        self.v1 = out[1]
262        self.v2 = int.from_bytes(out[2:4], 'big')
263
264        # Build the 3x3 matrix
265        matrix = np.zeros((3, 3))
266        for i in range(3):
267            for j in range(3):
268                k = i*3 + j
269                matrix[i, j] = self._read_ieee754(out[k*4 + 4 : k*4 + 8])
270
271        self.calibration_matrix = matrix
272
273    def _setup_measurement(self):
274        """
275        Retrieve s1, s2, s3 from the device, used in calibrate() and measure() calls.
276        """
277        # 0xc3 sets up measurement for the device
278        payload = [0xc3, 0x29, 0x27, 0x00, 0x01, self.v1]
279        out = self._bulk(payload, 15)
280
281        self.s1 = out[5]
282        self.s2 = out[6:10]
283        self.s3 = out[10:14]
284
285    def calibrate(self):
286        """
287        Perform black (dark) calibration.
288        The user should cover the SpyderX lens before calling this.
289        """
290        # Standard control transfer first
291        self._ctrl(0x41, 2, 2, 0)
292
293        # build payload from v2, s1, s2
294        payload = [self.v2 >> 8, self.v2 & 0xFF, self.s1] + list(self.s2)
295        out = self._bulk([0xd2, 0x3f, 0xb9, 0x00, 0x07] + payload, 13)
296
297        # parse the black calibration data
298        raw = struct.unpack('>HHHH', out[5:])
299        self.black_cal = np.array(raw[:3])
300
301    def measure(self):
302        """
303        Perform a measurement, returning XYZ as a numpy array [X, Y, Z].
304        Y is luminance in cd/m².
305        """
306        # Control transfer
307        self._ctrl(0x41, 2, 2, 0)
308
309        # build measurement payload
310        payload = [self.v2 >> 8, self.v2 & 0xFF, self.s1] + list(self.s2)
311        out = self._bulk([0xd2, 0x3f, 0xb9, 0x00, 0x07] + payload, 13)
312
313        raw = np.array(struct.unpack('>HHHH', out[5:]))[:3]
314        corrected = raw - self.black_cal
315        xyz = np.dot(corrected, self.calibration_matrix)
316        return xyz
317
318    def get_luminance(self):
319        """
320        Return just the Y (luminance) component in cd/m².
321        """
322        return self.measure()[1]
323
324    def get_rgb_luminance(self):
325        """
326        Convert measured XYZ to approximate linear RGB intensities.
327        Useful for relative channel brightness checks.
328        """
329        xyz = self.measure()
330        xyz_to_rgb = np.array([
331            [ 3.2406, -1.5372, -0.4986],
332            [-0.9689,  1.8758,  0.0415],
333            [ 0.0557, -0.2040,  1.0570]
334        ])
335        rgb = np.dot(xyz_to_rgb, xyz)
336        return tuple(rgb)

396    def close(self):
397        """
398        Release the interface, restore its kernel driver, and dispose resources.
399
400        This method is idempotent and safe after partially failed initialization.
401        """
402        if self._closed:
403            return
404        self._closed = True
405
406        if self.dev is None:
407            return
408
409        if self._interface_claimed:
410            try:
411                usb.util.release_interface(self.dev, self.INTERFACE)
412            except usb.core.USBError:
413                pass
414            finally:
415                self._interface_claimed = False
416
417        if self._detached_kernel_driver:
418            attach = getattr(self.dev, "attach_kernel_driver", None)
419            if callable(attach):
420                try:
421                    attach(self.INTERFACE)
422                except (NotImplementedError, usb.core.USBError):
423                    pass
424            self._detached_kernel_driver = False
425
426        try:
427            usb.util.dispose_resources(self.dev)
428        except usb.core.USBError:
429            pass

447class GammaFitter:
448    """
449    A class to fit a gamma function to luminance data for monitor calibration.
450
451    This class takes grayscale intensity values and corresponding luminance measurements,
452    normalizes the data, and fits a gamma function to estimate the display gamma.
453    It also provides tools for visualizing the fit and customizing fit parameters.
454
455    Attributes:
456        original_intensities (numpy.ndarray): Original grayscale intensity values.
457        original_luminance (numpy.ndarray): Original luminance measurements.
458        intensities (numpy.ndarray): Normalized grayscale intensities [0, 1].
459        luminance (numpy.ndarray): Normalized luminance values [0, 1].
460        params (list): Parameters of the fitted gamma function [a, b, c].
461        gamma (float): The gamma value (b) obtained from the fitted curve.
462        lower_bounds (list): Lower bounds for curve fitting parameters.
463        upper_bounds (list): Upper bounds for curve fitting parameters.
464        initial_guess (list): Initial guess for the curve fitting parameters.
465    """
466
467    def __init__(self, intensities, luminance):
468        """
469        Initializes the GammaFitter class.
470
471        Args:
472            intensities (list or array): The grayscale intensity values.
473            luminance (list or array): The corresponding luminance measurements.
474        """
475        # Store original values for plotting
476        self.original_intensities = np.array(intensities)
477        self.original_luminance = np.array(luminance)
478
479        # Normalize input data to the range [0, 1]
480        self.intensities = (self.original_intensities - np.min(self.original_intensities)) / (
481                    np.max(self.original_intensities) - np.min(self.original_intensities))
482        self.luminance = (self.original_luminance - np.min(self.original_luminance)) / (
483                    np.max(self.original_luminance) - np.min(self.original_luminance))
484
485        self.params = None
486        self.gamma = None
487
488        # Default values for bounds and initial guess based on monitor gamma calibration
489        self.lower_bounds = [0, 0, 0]  # Lower bounds: a ≥ 0, b ≥ 1 (physical gamma), c ≥ 0
490        self.upper_bounds = [10, 5, 1]  # Upper bounds: a ≤ 10, b ≤ 5, c ≤ 0.2
491        self.initial_guess = [0, 1, 0.01]  # Typical monitor gamma: a=1, b=2.2, c=0.01
492
493    def gamma_function(self, x, a, b, c):
494        """
495        Gamma function to model luminance as a function of intensity.
496
497        Args:
498            x (float or array): The normalized intensity values.
499            a (float): Scaling factor for luminance.
500            b (float): Gamma value (exponent).
501            c (float): Offset or baseline luminance.
502
503        Returns:
504            float or array: Modeled luminance values.
505        """
506        return a * (x ** b) + c
507
508    def set_bounds(self, lower_bounds, upper_bounds):
509        """
510        Set custom lower and upper bounds for the gamma fit parameters.
511
512        Args:
513            lower_bounds (list): Lower bounds for the parameters [a, b, c].
514            upper_bounds (list): Upper bounds for the parameters [a, b, c].
515        """
516        self.lower_bounds = lower_bounds
517        self.upper_bounds = upper_bounds
518
519    def set_initial_guess(self, initial_guess):
520        """
521        Set a custom initial guess for the gamma fit parameters.
522
523        Args:
524            initial_guess (list): Initial guess for the parameters [a, b, c].
525        """
526        self.initial_guess = initial_guess
527
528    def fit(self):
529        """
530        Fits the gamma function to the normalized intensity and luminance data.
531
532        Returns:
533            list: The fitted parameters [a, b, c], where 'b' is the gamma value.
534        """
535        bounds = (self.lower_bounds, self.upper_bounds)
536        self.params, _ = curve_fit(self.gamma_function, self.intensities, self.luminance, p0=self.initial_guess,
537                                   bounds=bounds)
538        self.gamma = self.params[1]
539        return self.params
540
541    def plot(self):
542        """
543        Plots the original data and the fitted gamma curve.
544
545        Raises:
546            ValueError: If the fit has not been performed before calling this method.
547        """
548        if self.params is None:
549            raise ValueError("Fit the data first using the 'fit' method before plotting.")
550
551        plt.figure(figsize=(8, 6))
552        plt.scatter(self.original_intensities, self.original_luminance, label='Original Data', color='blue', alpha=0.6)
553
554        # Generate the fit curve in the normalized range
555        x_fit_normalized = np.linspace(0, 1, 100)
556        y_fit_normalized = self.gamma_function(x_fit_normalized, *self.params)
557
558        # Convert back to the original scale for plotting
559        x_fit_original = x_fit_normalized * (
560                    np.max(self.original_intensities) - np.min(self.original_intensities)) + np.min(
561            self.original_intensities)
562        y_fit_original = y_fit_normalized * (
563                    np.max(self.original_luminance) - np.min(self.original_luminance)) + np.min(self.original_luminance)
564
565        plt.plot(x_fit_original, y_fit_original,
566                 label=f'Gamma={self.params[1]:.3f}, Base luminance={self.params[2]:.3f}',
567                 color='red')
568        plt.xlabel('Original Intensity')
569        plt.ylabel('Original Luminance')
570        plt.title('Gamma Function Fit for Monitor Calibration')
571        plt.legend()
572        plt.grid()
573        plt.show()

235    def _bulk(self, data, read_size, timeout=1000):
236        """
237        Helper for bulk transfers. Writes 'data' to endpoint 1, reads 'read_size' bytes from endpoint 0x81.

242    def _read_ieee754(self, b):
243        """
244        Decode a 32-bit IEEE-754 float from 4 bytes in 'b'.
245        The SpyderX data is little-endian, so we reverse and parse.
246        """
247        b = b[::-1]
248        raw = int.from_bytes(b, "big")
249        sign = (raw >> 31) & 1
250        exponent = (raw >> 23) & 0xFF
251        fraction = raw & 0x7FFFFF
252        return (-1) ** sign * (1 + fraction / 2**23) * 2**(exponent - 127)

254    def _get_calibration(self):
255        """
256        Retrieve factory calibration data, storing:
257          - v1, v2: calibration parameters
258          - calibration_matrix: 3x3 for XYZ computation
259        """
260        # Command 0xcb for factory calibration, read 47 bytes
261        out = self._bulk([0xcb, 0x05, 0x73, 0x00, 0x01, 0x00], 47)[5:]
262        self.v1 = out[1]
263        self.v2 = int.from_bytes(out[2:4], 'big')
264
265        # Build the 3x3 matrix
266        matrix = np.zeros((3, 3))
267        for i in range(3):
268            for j in range(3):
269                k = i*3 + j
270                matrix[i, j] = self._read_ieee754(out[k*4 + 4 : k*4 + 8])
